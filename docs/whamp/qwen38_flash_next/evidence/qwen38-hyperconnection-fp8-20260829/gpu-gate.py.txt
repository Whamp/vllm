import json
import statistics
from pathlib import Path
from types import SimpleNamespace

import torch
import torch.nn.functional as F
from safetensors import safe_open

import vllm.model_executor.layers.quantization.online.fp8 as online_fp8
from vllm.config import VllmConfig, set_current_vllm_config
from vllm.model_executor.kernels.linear import init_fp8_linear_kernel
from vllm.model_executor.layers.quantization.online.fp8 import (
    Fp8PerBlockOnlineLinearMethod,
)

MODEL = Path("/model")
INDEX = MODEL / "model.safetensors.index.json"
NUMERICAL_MAX_NRMSE = 0.04
NUMERICAL_MIN_COSINE = 0.999
PERFORMANCE_MIN_RATIO = 0.90


def load_layer_zero_weights():
    index = json.loads(INDEX.read_text())
    base = "model.language_model.layers.0.attn_hyper_connection."
    names = {
        "down": base + "input_mix_weight_down.weight",
        "inject": base + "block_inject_weight.weight",
        "up": base + "input_mix_weight_up.weight",
    }
    shards = {index["weight_map"][name] for name in names.values()}
    if len(shards) != 1:
        raise RuntimeError("Layer-0 hyperconnection tensors cross source shards")
    with safe_open(MODEL / next(iter(shards)), framework="pt", device="cpu") as src:
        tensors = {kind: src.get_tensor(name) for kind, name in names.items()}
    logical_rows = tensors["down"].shape[0] + tensors["inject"].shape[0]
    pad_rows = (-logical_rows) % 16
    merged_down = torch.cat(
        [
            tensors["down"],
            tensors["inject"],
            torch.zeros(
                (pad_rows, tensors["down"].shape[1]), dtype=tensors["down"].dtype
            ),
        ]
    )
    return {"merged_down": merged_down, "up": tensors["up"]}


def make_fp8_marlin_method(weight):
    online_fp8.get_current_vllm_config = lambda: SimpleNamespace(
        model_config=SimpleNamespace(dtype=torch.bfloat16)
    )
    with set_current_vllm_config(VllmConfig()):
        method = Fp8PerBlockOnlineLinearMethod()
        method.fp8_linear = init_fp8_linear_kernel(
            activation_quant_key=method.activation_quant_key,
            weight_quant_key=method.weight_quant_key,
            weight_shape=weight.shape,
            input_dtype=torch.bfloat16,
            out_dtype=torch.bfloat16,
            module_name="Qwen4ExpHyperconnectionFp8GpuGate",
        )
    layer = torch.nn.Module()
    layer.input_size_per_partition = weight.shape[1]
    layer.output_size_per_partition = weight.shape[0]
    layer.orig_dtype = torch.bfloat16
    layer.weight_block_size = [128, 128]
    layer.register_parameter(
        "weight", torch.nn.Parameter(weight.cuda(), requires_grad=False)
    )
    method.process_weights_after_loading(layer)
    if type(method.fp8_linear).__name__ != "MarlinFP8ScaledMMLinearKernel":
        raise RuntimeError(
            f"Expected Marlin FP8 dispatch, got {type(method.fp8_linear).__name__}"
        )
    return method, layer


def output_metrics(reference, candidate):
    reference = reference.float().reshape(-1)
    candidate = candidate.float().reshape(-1)
    error = candidate - reference
    nrmse = torch.linalg.vector_norm(error) / torch.linalg.vector_norm(reference)
    cosine = F.cosine_similarity(reference.view(1, -1), candidate.view(1, -1))
    return {
        "normalized_rmse": float(nrmse),
        "cosine_similarity": float(cosine),
        "maximum_absolute_error": float(error.abs().amax()),
    }


def time_operation(operation, warmups, iterations, repeats):
    for _ in range(warmups):
        operation()
    torch.cuda.synchronize()
    samples = []
    for _ in range(repeats):
        start = torch.cuda.Event(enable_timing=True)
        end = torch.cuda.Event(enable_timing=True)
        start.record()
        for _ in range(iterations):
            operation()
        end.record()
        end.synchronize()
        samples.append(start.elapsed_time(end) * 1000.0 / iterations)
    return {"median_microseconds": statistics.median(samples), "samples": samples}


def graph_determinism(method, layer, shape):
    static_input = torch.randn(shape, device="cuda", dtype=torch.bfloat16)
    stream = torch.cuda.Stream()
    with torch.cuda.stream(stream):
        for _ in range(3):
            method.apply(layer, static_input)
    stream.synchronize()
    graph = torch.cuda.CUDAGraph()
    with torch.cuda.graph(graph):
        static_output = method.apply(layer, static_input)
    first_input = torch.randn_like(static_input)
    static_input.copy_(first_input)
    graph.replay()
    first_output = static_output.clone()
    static_input.copy_(first_input)
    graph.replay()
    second_output = static_output.clone()
    return {
        "bitwise_equal": bool(torch.equal(first_output, second_output)),
        "finite": bool(torch.isfinite(first_output).all()),
    }


def main():
    if not torch.cuda.is_available():
        raise RuntimeError("CUDA is required")
    if torch.cuda.get_device_capability() != (8, 6):
        raise RuntimeError(f"Expected SM86, got {torch.cuda.get_device_capability()}")
    torch.manual_seed(20260829)
    weights = load_layer_zero_weights()
    results = {
        "schema_version": 1,
        "device_name": torch.cuda.get_device_name(),
        "compute_capability": list(torch.cuda.get_device_capability()),
        "acceptance": {
            "maximum_normalized_rmse": NUMERICAL_MAX_NRMSE,
            "minimum_cosine_similarity": NUMERICAL_MIN_COSINE,
            "minimum_fp8_to_bf16_speed_ratio": PERFORMANCE_MIN_RATIO,
        },
        "matrices": {},
    }
    methods = {}
    for matrix_name, cpu_weight in weights.items():
        method, layer = make_fp8_marlin_method(cpu_weight)
        methods[matrix_name] = (method, layer, cpu_weight.cuda())
        matrix_result = {
            "logical_shape": list(cpu_weight.shape),
            "packed_weight_bytes": layer.weight.numel() * layer.weight.element_size(),
            "scale_bytes": layer.weight_scale_inv.numel()
            * layer.weight_scale_inv.element_size(),
            "workspace_bytes": layer.workspace.numel() * layer.workspace.element_size(),
            "kernel": type(method.fp8_linear).__name__,
            "shapes": {},
        }
        for tokens in (1, 2, 256):
            x = torch.randn(
                (tokens, cpu_weight.shape[1]), device="cuda", dtype=torch.bfloat16
            )
            reference = F.linear(x, methods[matrix_name][2])
            candidate = method.apply(layer, x)
            shape_result = output_metrics(reference, candidate)
            if shape_result["normalized_rmse"] > NUMERICAL_MAX_NRMSE:
                raise RuntimeError(
                    f"{matrix_name} M={tokens} NRMSE gate failed: {shape_result}"
                )
            if shape_result["cosine_similarity"] < NUMERICAL_MIN_COSINE:
                raise RuntimeError(
                    f"{matrix_name} M={tokens} cosine gate failed: {shape_result}"
                )
            iterations = 1000 if tokens <= 2 else 100
            reference_weight = methods[matrix_name][2]

            def bf16_operation(input_tensor=x, weight=reference_weight):
                return F.linear(input_tensor, weight)

            def fp8_operation(input_tensor=x, linear_method=method, fp8_layer=layer):
                return linear_method.apply(fp8_layer, input_tensor)

            reference_timing = time_operation(bf16_operation, 20, iterations, 5)
            candidate_timing = time_operation(fp8_operation, 20, iterations, 5)
            shape_result["bf16_timing"] = reference_timing
            shape_result["fp8_marlin_timing"] = candidate_timing
            shape_result["speed_ratio"] = (
                reference_timing["median_microseconds"]
                / candidate_timing["median_microseconds"]
            )
            matrix_result["shapes"][str(tokens)] = shape_result
        matrix_result["cuda_graph"] = graph_determinism(
            method, layer, (2, cpu_weight.shape[1])
        )
        if not all(matrix_result["cuda_graph"].values()):
            raise RuntimeError(
                f"{matrix_name} CUDA Graph gate failed: {matrix_result['cuda_graph']}"
            )
        results["matrices"][matrix_name] = matrix_result

    results["performance_gate_passed"] = all(
        shape["speed_ratio"] >= PERFORMANCE_MIN_RATIO
        for matrix in results["matrices"].values()
        for shape in matrix["shapes"].values()
    )
    print(json.dumps(results, indent=2, sort_keys=True))


if __name__ == "__main__":
    main()
