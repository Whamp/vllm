"""Bounded RTX 3090 gate for Qwen3.8 QSA per-token-head INT8 cache."""

from __future__ import annotations

import argparse
import json
import statistics
from pathlib import Path

import torch
from vllm.models.qwen3_8_flash_next.nvidia import (
    model as _qwen38_model,  # noqa: F401
)
from vllm.models.qwen3_8_flash_next.nvidia.ops.qsa import (
    qsa_sparse_paged_attention,
)
from vllm.models.qwen3_8_flash_next.nvidia.qsa import (
    Qwen3_8FlashNextQSAFlashAttentionImpl,
)

from vllm.v1.attention.backend import AttentionType
from vllm.v1.attention.ops.triton_reshape_and_cache_flash import (
    triton_reshape_and_cache_flash,
)
from vllm.v1.kv_cache_interface import KVQuantMode

MAX_NRMSE = 0.02
MIN_COSINE = 0.999
MAX_READER_RATIO = 1.25
HEAD_DIM = 256
NUM_QUERY_HEADS = 6
NUM_KV_HEADS = 1
PAGE_SIZE = 16
TOPK = 2051


def int8_impl() -> Qwen3_8FlashNextQSAFlashAttentionImpl:
    impl = Qwen3_8FlashNextQSAFlashAttentionImpl.__new__(
        Qwen3_8FlashNextQSAFlashAttentionImpl
    )
    impl.attn_type = AttentionType.DECODER
    impl._kv_quant_mode = KVQuantMode.INT8_PER_TOKEN_HEAD
    impl._is_per_token_head_quant = True
    impl._k_scale_cache = None
    impl._v_scale_cache = None
    return impl


def allocate_int8_cache(num_blocks: int):
    scale_pad = torch.tensor([], dtype=torch.float32).element_size()
    padded_head_dim = HEAD_DIM + scale_pad
    raw = torch.zeros(
        num_blocks,
        NUM_KV_HEADS,
        PAGE_SIZE,
        2 * padded_head_dim,
        device="cuda",
        dtype=torch.int8,
    )
    impl = int8_impl()
    key_padded, value_padded = impl._pth_key_value_caches(raw)
    return (
        impl,
        raw,
        key_padded[..., :HEAD_DIM],
        value_padded[..., :HEAD_DIM],
        impl._k_scale_cache,
        impl._v_scale_cache,
    )


def quantize_reference(tensor: torch.Tensor):
    scales = torch.clamp(tensor.float().abs().amax(dim=-1) / 127.0, min=1e-6)
    scaled = tensor.float() / scales.unsqueeze(-1)
    rounded = torch.where(scaled >= 0, scaled + 0.5, scaled - 0.5)
    return rounded.clamp(-128, 127).to(torch.int8), scales


def output_metrics(reference: torch.Tensor, actual: torch.Tensor):
    reference = reference.float()
    actual = actual.float()
    delta = actual - reference
    nrmse = torch.sqrt(torch.mean(delta.square())) / torch.sqrt(
        torch.mean(reference.square())
    )
    cosine = torch.nn.functional.cosine_similarity(
        reference.flatten(), actual.flatten(), dim=0
    )
    return {
        "normalized_rmse": float(nrmse),
        "cosine_similarity": float(cosine),
        "maximum_absolute_error": float(delta.abs().max()),
    }


def torch_reference(
    query,
    key_cache,
    value_cache,
    logical_indices,
    block_table,
    token_to_req,
):
    output = torch.zeros_like(query)
    repeats = query.shape[1] // key_cache.shape[2]
    for row in range(query.shape[0]):
        logical = logical_indices[row]
        logical = logical[logical >= 0].long()
        if not logical.numel():
            continue
        request = token_to_req[row].long()
        pages = block_table[request, logical // PAGE_SIZE].long()
        offsets = logical % PAGE_SIZE
        keys = key_cache[pages, offsets].repeat_interleave(repeats, dim=1)
        values = value_cache[pages, offsets].repeat_interleave(repeats, dim=1)
        scores = torch.einsum("hd,khd->hk", query[row].float(), keys.float())
        probabilities = torch.softmax(scores * HEAD_DIM**-0.5, dim=-1)
        output[row] = torch.einsum("hk,khd->hd", probabilities, values.float()).to(
            query.dtype
        )
    return output


def writer_and_numerical_gate():
    num_blocks = 2
    num_tokens = num_blocks * PAGE_SIZE
    slots = torch.arange(num_tokens, device="cuda", dtype=torch.int64)

    exact_codes = (
        torch.arange(HEAD_DIM, device="cuda", dtype=torch.int32) % 255 - 127
    ).to(torch.bfloat16)
    exact_key = exact_codes.view(1, 1, -1).expand(num_tokens, 1, -1).contiguous()
    exact_value = -exact_key
    exact_impl, exact_raw, exact_key_cache, exact_value_cache, _, _ = (
        allocate_int8_cache(num_blocks)
    )
    exact_impl.do_kv_cache_update(None, exact_key, exact_value, exact_raw, slots)
    expected_key, _ = quantize_reference(exact_key)
    expected_value, _ = quantize_reference(exact_value)
    torch.testing.assert_close(
        exact_key_cache.reshape_as(expected_key), expected_key, rtol=0, atol=0
    )
    torch.testing.assert_close(
        exact_value_cache.reshape_as(expected_value), expected_value, rtol=0, atol=0
    )

    torch.manual_seed(311)
    key = torch.randn(
        num_tokens,
        NUM_KV_HEADS,
        HEAD_DIM,
        device="cuda",
        dtype=torch.bfloat16,
    )
    value = torch.randn_like(key)
    impl, raw, key_cache, value_cache, key_scales, value_scales = allocate_int8_cache(
        num_blocks
    )
    impl.do_kv_cache_update(None, key, value, raw, slots)
    _, expected_key_scales = quantize_reference(key)
    _, expected_value_scales = quantize_reference(value)
    torch.testing.assert_close(
        key_scales.reshape_as(expected_key_scales), expected_key_scales
    )
    torch.testing.assert_close(
        value_scales.reshape_as(expected_value_scales), expected_value_scales
    )

    query = torch.randn(
        1,
        NUM_QUERY_HEADS,
        HEAD_DIM,
        device="cuda",
        dtype=torch.bfloat16,
    )
    indices = torch.arange(num_tokens, device="cuda", dtype=torch.int32)[None]
    block_table = torch.arange(num_blocks, device="cuda", dtype=torch.int32)[None]
    token_to_req = torch.zeros(1, device="cuda", dtype=torch.int32)
    actual = qsa_sparse_paged_attention(
        query,
        key_cache,
        value_cache,
        indices,
        block_table,
        token_to_req,
        kv_quant_mode=KVQuantMode.INT8_PER_TOKEN_HEAD,
        key_scale_cache=key_scales,
        value_scale_cache=value_scales,
    )
    dequantized_key = key_cache.float() * key_scales.unsqueeze(-1)
    dequantized_value = value_cache.float() * value_scales.unsqueeze(-1)
    reference = torch_reference(
        query,
        dequantized_key,
        dequantized_value,
        indices,
        block_table,
        token_to_req,
    )
    torch.testing.assert_close(actual, reference, rtol=2e-2, atol=2e-2)
    bf16_reference = torch_reference(
        query,
        key.reshape(num_blocks, PAGE_SIZE, NUM_KV_HEADS, HEAD_DIM),
        value.reshape(num_blocks, PAGE_SIZE, NUM_KV_HEADS, HEAD_DIM),
        indices,
        block_table,
        token_to_req,
    )
    metrics = output_metrics(bf16_reference, actual)
    if metrics["normalized_rmse"] > MAX_NRMSE:
        raise RuntimeError(f"INT8 QSA NRMSE gate failed: {metrics}")
    if metrics["cosine_similarity"] < MIN_COSINE:
        raise RuntimeError(f"INT8 QSA cosine gate failed: {metrics}")
    return {
        "writer_codes_match": True,
        "writer_scales_match": True,
        "output": metrics,
    }


def time_operation(operation, iterations=30, repeats=5):
    for _ in range(10):
        operation()
    torch.accelerator.synchronize()
    samples = []
    for _ in range(repeats):
        start = torch.cuda.Event(enable_timing=True)
        end = torch.cuda.Event(enable_timing=True)
        start.record()
        for _ in range(iterations):
            operation()
        end.record()
        end.synchronize()
        samples.append(start.elapsed_time(end) * 1000 / iterations)
    return {
        "median_microseconds": statistics.median(samples),
        "samples": samples,
    }


def performance_case(rows: int):
    torch.manual_seed(410 + rows)
    num_blocks = (TOPK + PAGE_SIZE - 1) // PAGE_SIZE
    num_tokens = num_blocks * PAGE_SIZE
    key = torch.randn(
        num_tokens,
        NUM_KV_HEADS,
        HEAD_DIM,
        device="cuda",
        dtype=torch.bfloat16,
    )
    value = torch.randn_like(key)
    slots = torch.arange(num_tokens, device="cuda", dtype=torch.int64)
    impl, raw, int8_key, int8_value, key_scales, value_scales = allocate_int8_cache(
        num_blocks
    )
    impl.do_kv_cache_update(None, key, value, raw, slots)
    bf16_key = key.reshape(num_blocks, PAGE_SIZE, NUM_KV_HEADS, HEAD_DIM).contiguous()
    bf16_value = value.reshape_as(bf16_key).contiguous()
    query = torch.randn(
        rows,
        NUM_QUERY_HEADS,
        HEAD_DIM,
        device="cuda",
        dtype=torch.bfloat16,
    )
    indices = torch.arange(TOPK, device="cuda", dtype=torch.int32).expand(rows, -1)
    block_table = torch.arange(num_blocks, device="cuda", dtype=torch.int32)[None]
    token_to_req = torch.zeros(rows, device="cuda", dtype=torch.int32)

    def bf16_reader():
        return qsa_sparse_paged_attention(
            query,
            bf16_key,
            bf16_value,
            indices,
            block_table,
            token_to_req,
        )

    def int8_reader():
        return qsa_sparse_paged_attention(
            query,
            int8_key,
            int8_value,
            indices,
            block_table,
            token_to_req,
            kv_quant_mode=KVQuantMode.INT8_PER_TOKEN_HEAD,
            key_scale_cache=key_scales,
            value_scale_cache=value_scales,
        )

    bf16_output = bf16_reader()
    int8_output = int8_reader()
    metrics = output_metrics(bf16_output, int8_output)
    bf16_timing = time_operation(bf16_reader)
    int8_timing = time_operation(int8_reader)
    reader_ratio = (
        int8_timing["median_microseconds"] / bf16_timing["median_microseconds"]
    )

    bf16_write_cache = torch.empty_like(bf16_key)
    bf16_value_cache = torch.empty_like(bf16_value)
    scalar_scale = torch.ones(1, device="cuda", dtype=torch.float32)

    def bf16_writer():
        triton_reshape_and_cache_flash(
            key,
            value,
            bf16_write_cache,
            bf16_value_cache,
            slots,
            "auto",
            scalar_scale,
            scalar_scale,
        )

    def int8_writer():
        impl.do_kv_cache_update(None, key, value, raw, slots)

    bf16_writer_timing = time_operation(bf16_writer)
    int8_writer_timing = time_operation(int8_writer)
    return {
        "rows": rows,
        "output": metrics,
        "reader": {
            "bf16": bf16_timing,
            "int8": int8_timing,
            "int8_to_bf16_ratio": reader_ratio,
            "passed": reader_ratio <= MAX_READER_RATIO,
        },
        "writer": {
            "bf16": bf16_writer_timing,
            "int8": int8_writer_timing,
            "int8_to_bf16_ratio": int8_writer_timing["median_microseconds"]
            / bf16_writer_timing["median_microseconds"],
        },
    }


def graph_gate():
    torch.manual_seed(701)
    num_blocks = 4
    num_tokens = num_blocks * PAGE_SIZE
    key = torch.randn(
        num_tokens,
        NUM_KV_HEADS,
        HEAD_DIM,
        device="cuda",
        dtype=torch.bfloat16,
    )
    value = torch.randn_like(key)
    query = torch.randn(
        1,
        NUM_QUERY_HEADS,
        HEAD_DIM,
        device="cuda",
        dtype=torch.bfloat16,
    )
    slots = torch.arange(num_tokens, device="cuda", dtype=torch.int64)
    indices = torch.arange(num_tokens, device="cuda", dtype=torch.int32)[None]
    block_table = torch.arange(num_blocks, device="cuda", dtype=torch.int32)[None]
    token_to_req = torch.zeros(1, device="cuda", dtype=torch.int32)
    impl, raw, key_cache, value_cache, key_scales, value_scales = allocate_int8_cache(
        num_blocks
    )

    def transaction():
        impl.do_kv_cache_update(None, key, value, raw, slots)
        return qsa_sparse_paged_attention(
            query,
            key_cache,
            value_cache,
            indices,
            block_table,
            token_to_req,
            kv_quant_mode=KVQuantMode.INT8_PER_TOKEN_HEAD,
            key_scale_cache=key_scales,
            value_scale_cache=value_scales,
        )

    for _ in range(3):
        transaction()
    graph = torch.cuda.CUDAGraph()
    with torch.cuda.graph(graph):
        output = transaction()
    graph.replay()
    first = output.clone()
    graph.replay()
    second = output.clone()
    if not torch.equal(first, second) or not torch.isfinite(first).all():
        raise RuntimeError("INT8 QSA CUDA Graph replay is not deterministic and finite")
    return {"bitwise_equal": True, "finite": True}


def main():
    parser = argparse.ArgumentParser()
    parser.add_argument("--output", type=Path, required=True)
    args = parser.parse_args()
    assert torch.cuda.device_count() == 1
    capability = torch.cuda.get_device_capability()
    if capability != (8, 6):
        raise RuntimeError(f"Expected SM86, got {capability}")
    numerical = writer_and_numerical_gate()
    graph = graph_gate()
    performance = {"m1": performance_case(1), "m256": performance_case(256)}
    result = {
        "schema_version": 1,
        "device": torch.cuda.get_device_name(),
        "compute_capability": list(capability),
        "acceptance": {
            "maximum_normalized_rmse": MAX_NRMSE,
            "minimum_cosine_similarity": MIN_COSINE,
            "maximum_int8_to_bf16_reader_time_ratio": MAX_READER_RATIO,
        },
        "numerical": numerical,
        "cuda_graph": graph,
        "performance": performance,
        "performance_gate_passed": all(
            case["reader"]["passed"] for case in performance.values()
        ),
    }
    args.output.parent.mkdir(parents=True, exist_ok=True)
    args.output.write_text(json.dumps(result, indent=2, sort_keys=True) + "\n")
    print(json.dumps(result, indent=2, sort_keys=True))


if __name__ == "__main__":
    main()
