from __future__ import annotations

import torch
from hypothesis import event, given, settings
from hypothesis import strategies as st

from vllm.models.qwen4_exp.nvidia import (
    model as _qwen4_exp_model,  # noqa: F401
)
from vllm.models.qwen4_exp.nvidia import qsa as qsa_backend
from vllm.v1.kv_cache_interface import FullAttentionSpec, KVQuantMode


@settings(max_examples=100, deadline=None)
@given(
    num_blocks=st.integers(min_value=1, max_value=4),
    num_kv_heads=st.integers(min_value=1, max_value=4),
    block_size=st.sampled_from([16, 32, 64]),
    head_size=st.sampled_from([16, 32, 64, 128, 256]),
    physical_layout=st.sampled_from(["HND", "NHD"]),
    scale_offset=st.integers(min_value=-31, max_value=31),
)
def test_qsa_int8_inline_scale_views_follow_declared_cache_layout(
    num_blocks: int,
    num_kv_heads: int,
    block_size: int,
    head_size: int,
    physical_layout: str,
    scale_offset: int,
) -> None:
    """INT8 scale views must address only each K/V row's four-byte tail."""
    event(f"layout={physical_layout}")
    event(f"head_size={head_size}")

    spec = FullAttentionSpec(
        block_size=block_size,
        num_kv_heads=num_kv_heads,
        head_size=head_size,
        head_size_v=head_size,
        dtype=torch.int8,
        kv_quant_mode=KVQuantMode.INT8_PER_TOKEN_HEAD,
    )
    customized = qsa_backend.Qwen4ExpQSAFlashAttentionBackend.customize_spec(spec)
    expected_padded_head_size = head_size + 4
    expected_content_bytes = 2 * expected_padded_head_size

    assert customized.state_content_bytes == expected_content_bytes
    logical_shape = qsa_backend.Qwen4ExpQSAFlashAttentionBackend.get_kv_cache_shape(
        num_blocks,
        block_size,
        num_kv_heads,
        head_size,
        "int8_per_token_head",
    )
    assert logical_shape == (
        num_blocks,
        num_kv_heads,
        block_size,
        expected_content_bytes,
    )

    if physical_layout == "HND":
        raw_cache = torch.zeros(logical_shape, dtype=torch.int8)
    else:
        physical = torch.zeros(
            num_blocks,
            block_size,
            num_kv_heads,
            expected_content_bytes,
            dtype=torch.int8,
        )
        raw_cache = physical.permute(0, 2, 1, 3)

    impl = qsa_backend.Qwen4ExpQSAFlashAttentionImpl.__new__(
        qsa_backend.Qwen4ExpQSAFlashAttentionImpl
    )
    impl._kv_quant_mode = KVQuantMode.INT8_PER_TOKEN_HEAD
    impl._is_per_token_head_quant = True
    impl._k_scale_cache = None
    impl._v_scale_cache = None
    key_padded, value_padded = impl._pth_key_value_caches(raw_cache)
    key_scales = impl._k_scale_cache
    value_scales = impl._v_scale_cache
    assert key_scales is not None and value_scales is not None

    scale_values = (
        torch.arange(
            num_blocks * block_size * num_kv_heads,
            dtype=torch.float32,
        ).reshape(num_blocks, block_size, num_kv_heads)
        + scale_offset
        + 0.25
    )
    key_scales.copy_(scale_values)
    value_scales.copy_(-scale_values)

    key_scale_bytes = raw_cache[..., head_size:expected_padded_head_size].view(
        torch.float32
    )
    value_scale_bytes = raw_cache[
        ...,
        expected_padded_head_size + head_size : 2 * expected_padded_head_size,
    ].view(torch.float32)
    torch.testing.assert_close(
        key_scale_bytes.squeeze(-1), scale_values.permute(0, 2, 1)
    )
    torch.testing.assert_close(
        value_scale_bytes.squeeze(-1), -scale_values.permute(0, 2, 1)
    )

    key_padded[..., :head_size].fill_(7)
    value_padded[..., :head_size].fill_(-7)
    torch.testing.assert_close(key_scales, scale_values)
    torch.testing.assert_close(value_scales, -scale_values)
