@settings(max_examples=150, deadline=None)
@given(
    num_blocks=st.integers(min_value=1, max_value=4),
    num_kv_heads=st.integers(min_value=1, max_value=4),
    block_size=st.sampled_from([16, 32, 64]),
    head_size=st.sampled_from([16, 32, 64, 128, 256]),
    physical_layout=st.sampled_from(["HND", "NHD"]),
    scale_offset=st.integers(min_value=-31, max_value=31),
)
def test_qsa_int4_inline_scale_views_follow_declared_cache_layout(
    num_blocks: int,
    num_kv_heads: int,
    block_size: int,
    head_size: int,
    physical_layout: str,
    scale_offset: int,
) -> None:
    """Packed INT4 scale views must address only each K/V four-byte tail."""
    event(f"layout={physical_layout}")
    event(f"head_size={head_size}")

    packed_head_size = head_size // 2
    padded_packed_head_size = packed_head_size + 4
    expected_content_bytes = 2 * padded_packed_head_size
    spec = FullAttentionSpec(
        block_size=block_size,
        num_kv_heads=num_kv_heads,
        head_size=head_size,
        head_size_v=head_size,
        dtype=torch.uint8,
        kv_quant_mode=KVQuantMode.INT4_PER_TOKEN_HEAD,
    )
    customized = qsa_backend.Qwen4ExpQSAFlashAttentionBackend.customize_spec(spec)

    assert "int4_per_token_head" in (
        qsa_backend.Qwen4ExpQSAFlashAttentionBackend.supported_kv_cache_dtypes
    )
    assert customized.state_content_bytes == expected_content_bytes
    logical_shape = qsa_backend.Qwen4ExpQSAFlashAttentionBackend.get_kv_cache_shape(
        num_blocks,
        block_size,
        num_kv_heads,
        head_size,
        "int4_per_token_head",
    )
    assert logical_shape == (
        num_blocks,
        num_kv_heads,
        block_size,
        expected_content_bytes,
    )

    if physical_layout == "HND":
        raw_cache = torch.zeros(logical_shape, dtype=torch.uint8)
    else:
        physical = torch.zeros(
            num_blocks,
            block_size,
            num_kv_heads,
            expected_content_bytes,
            dtype=torch.uint8,
        )
        raw_cache = physical.permute(0, 2, 1, 3)

    impl = qsa_backend.Qwen4ExpQSAFlashAttentionImpl.__new__(
        qsa_backend.Qwen4ExpQSAFlashAttentionImpl
    )
    impl.head_size = head_size
    impl._kv_quant_mode = KVQuantMode.INT4_PER_TOKEN_HEAD
    key_padded, value_padded, key_scales, value_scales = (
        impl.per_token_head_cache_views(raw_cache)
    )
    assert key_padded.shape == (
        num_blocks,
        block_size,
        num_kv_heads,
        packed_head_size,
    )
    assert value_padded.shape == key_padded.shape

    scale_values = (
        torch.arange(
            num_blocks * block_size * num_kv_heads,
            dtype=torch.float32,
        ).reshape(num_blocks, block_size, num_kv_heads)
        + scale_offset
        + 0.25
    )
    key_scales.copy_(scale_values)
    value_scales.copy_(-scale_values)

    key_scale_bytes = raw_cache[..., packed_head_size:padded_packed_head_size].view(
        torch.float32
    )
    value_scale_bytes = raw_cache[
        ...,
        padded_packed_head_size + packed_head_size : 2 * padded_packed_head_size,
    ].view(torch.float32)
    torch.testing.assert_close(
        key_scale_bytes.squeeze(-1), scale_values.permute(0, 2, 1)
    )
    torch.testing.assert_close(
        value_scale_bytes.squeeze(-1), -scale_values.permute(0, 2, 1)
    )

    key_padded[..., :packed_head_size].fill_(0xA5)
    value_padded[..., :packed_head_size].fill_(0x5A)
    torch.testing.assert_close(key_scales, scale_values)
    torch.testing.assert_close(value_scales, -scale_values)


@settings(max_examples=100, deadline=None)
@given(
    head_dim=st.sampled_from([16, 32, 64, 128, 256]),
    num_rows=st.integers(min_value=1, max_value=8),
    seed=st.integers(min_value=0, max_value=2**32 - 1),
)
def test_single_rht_matrix_pair_round_trips_generated_rows(
    head_dim: int,
    num_rows: int,
    seed: int,
) -> None:
    """The cached matrix pair must preserve rows after inverse normalization."""
    generator = torch.Generator(device="cpu")
    generator.manual_seed(seed)
    rows = torch.randn(
        num_rows,
        head_dim,
        generator=generator,
        dtype=torch.bfloat16,
    )
    forward, inverse = get_single_rht_matrix_pair(
        head_dim, torch.bfloat16, torch.device("cpu")
    )
    cached_forward, cached_inverse = get_single_rht_matrix_pair(
        head_dim, torch.bfloat16, torch.device("cpu")
    )

    assert forward.shape == inverse.shape == (head_dim, head_dim)
    assert forward.data_ptr() == cached_forward.data_ptr()
    assert inverse.data_ptr() == cached_inverse.data_ptr()
    restored = torch.matmul(torch.matmul(rows, forward), inverse) / head_dim
    torch.testing.assert_close(restored, rows, rtol=2e-2, atol=2e-2)


@st.composite
def qsa_int4_sparse_cases(
    draw: st.DrawFn,
) -> tuple[int, int, int, int, list[list[int]]]:
    """Generate bounded sparse selections with duplicates and inert sentinels."""
    num_requests = draw(st.integers(min_value=1, max_value=2))
    pages_per_request = draw(st.integers(min_value=1, max_value=4))
    num_rows = draw(st.integers(min_value=1, max_value=4))
    selection_width = draw(
        st.integers(min_value=1, max_value=min(32, pages_per_request * 16))
    )
    max_logical_token = pages_per_request * 16 - 1
    selections = []
    for _ in range(num_rows):
        first = draw(st.integers(min_value=0, max_value=max_logical_token))
        rest = draw(
            st.lists(
                st.one_of(
                    st.just(-1),
                    st.integers(min_value=0, max_value=max_logical_token),
                ),
                min_size=selection_width - 1,
                max_size=selection_width - 1,
            )
        )
        selections.append([first, *rest])
    seed = draw(st.integers(min_value=0, max_value=2**32 - 1))
    return num_requests, pages_per_request, num_rows, seed, selections


def _qsa_bf16_reference(
    query: torch.Tensor,
    key_cache: torch.Tensor,
    value_cache: torch.Tensor,
    logical_indices: torch.Tensor,
    block_table: torch.Tensor,
    token_to_req: torch.Tensor,
) -> torch.Tensor:
    output = torch.zeros_like(query)
    repeats = query.shape[1] // key_cache.shape[2]
    page_size = key_cache.shape[1]
    for row in range(query.shape[0]):
        logical = logical_indices[row]
        logical = logical[logical >= 0].long()
        request = token_to_req[row].long()
        pages = block_table[request, logical // page_size].long()
        offsets = logical % page_size
        keys = key_cache[pages, offsets].repeat_interleave(repeats, dim=1)
        values = value_cache[pages, offsets].repeat_interleave(repeats, dim=1)
        scores = torch.einsum("hd,khd->hk", query[row].float(), keys.float())
        probabilities = torch.softmax(scores * query.shape[-1] ** -0.5, dim=-1)
        output[row] = torch.einsum("hk,khd->hd", probabilities, values.float()).to(
            query.dtype
        )
    return output


@requires_qsa_kernels
@settings(max_examples=30, deadline=None)
@given(case=qsa_int4_sparse_cases())
def test_qsa_int4_sparse_attention_tracks_bf16_reference(
    case: tuple[int, int, int, int, list[list[int]]],
) -> None:
    """Packed QSA stays close to BF16 across generated sparse selections."""
    num_requests, pages_per_request, num_rows, seed, selections = case
    event(f"requests={num_requests}")
    event(f"rows={num_rows}")
    event(f"has_sentinel={any(-1 in row for row in selections)}")
    event(
        "has_duplicate="
        + str(
            any(
                len({value for value in row if value >= 0}) < len(row)
                for row in selections
            )
        )
    )

    generator = torch.Generator(device="cuda")
    generator.manual_seed(seed)
    head_dim = 256
    num_query_heads = 6
    num_kv_heads = 1
    page_size = 16
    num_blocks = num_requests * pages_per_request
    num_tokens = num_blocks * page_size
    key = torch.randn(
        num_tokens,
        num_kv_heads,
        head_dim,
        generator=generator,
        device="cuda",
        dtype=torch.bfloat16,
    )
    value = torch.randn(
        num_tokens,
        num_kv_heads,
        head_dim,
        generator=generator,
        device="cuda",
        dtype=torch.bfloat16,
    )
    query = torch.randn(
        num_rows,
        num_query_heads,
        head_dim,
        generator=generator,
        device="cuda",
        dtype=torch.bfloat16,
    )

    raw_shape = qsa_backend.Qwen4ExpQSAFlashAttentionBackend.get_kv_cache_shape(
        num_blocks,
        page_size,
        num_kv_heads,
        head_dim,
        "int4_per_token_head",
    )
    raw_cache = torch.zeros(raw_shape, device="cuda", dtype=torch.uint8)
    impl = qsa_backend.Qwen4ExpQSAFlashAttentionImpl.__new__(
        qsa_backend.Qwen4ExpQSAFlashAttentionImpl
    )
    impl.head_size = head_dim
    impl._kv_quant_mode = KVQuantMode.INT4_PER_TOKEN_HEAD
    key_packed, value_packed, key_scales, value_scales = (
        impl.per_token_head_cache_views(raw_cache)
    )
    reshape_and_cache_int4(
        key,
        value,
        key_packed,
        value_packed,
        torch.arange(num_tokens, device="cuda", dtype=torch.int64),
        k_scale_cache=key_scales,
        v_scale_cache=value_scales,
    )

    block_table = torch.arange(num_blocks, device="cuda", dtype=torch.int32).reshape(
        num_requests, pages_per_request
    )
    token_to_req = torch.arange(num_rows, device="cuda", dtype=torch.int32)
    token_to_req %= num_requests
    logical_indices = torch.tensor(selections, device="cuda", dtype=torch.int32)
    actual = qsa_sparse_paged_attention(
        query,
        key_packed,
        value_packed,
        logical_indices,
        block_table,
        token_to_req,
        kv_quant_mode=KVQuantMode.INT4_PER_TOKEN_HEAD,
        key_scale_cache=key_scales,
        value_scale_cache=value_scales,
    )
    key_cache = key.reshape(num_blocks, page_size, num_kv_heads, head_dim)
    value_cache = value.reshape_as(key_cache)
    expected = _qsa_bf16_reference(
        query,
        key_cache,
        value_cache,
        logical_indices,
        block_table,
        token_to_req,
    )

    delta = actual.float() - expected.float()
    normalized_rmse = torch.sqrt(torch.mean(delta.square())) / torch.clamp(
        torch.sqrt(torch.mean(expected.float().square())), min=1e-8
    )
    cosine = torch.nn.functional.cosine_similarity(
        actual.float().flatten(), expected.float().flatten(), dim=0
    )
    target(float(normalized_rmse), label="normalized_rmse")
    assert torch.isfinite(actual).all()
    assert normalized_rmse <= 0.25
    assert cosine >= 0.95
