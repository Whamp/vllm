"""RTX 3090 acceptance gate for Qwen3.8 QSA per-token-head INT4 cache."""

from __future__ import annotations

import argparse
import json
import statistics
from pathlib import Path

import torch
from vllm.models.qwen3_8_flash_next.nvidia import (
    model as _qwen38_model,  # noqa: F401
)
from vllm.models.qwen3_8_flash_next.nvidia.ops.qsa import (
    qsa_sparse_paged_attention,
)
from vllm.models.qwen3_8_flash_next.nvidia.qsa import (
    Qwen3_8FlashNextQSAFlashAttentionBackend,
    Qwen3_8FlashNextQSAFlashAttentionImpl,
)

from vllm.v1.attention.ops.int4_per_token_head import reshape_and_cache_int4
from vllm.v1.attention.ops.triton_reshape_and_cache_flash import (
    triton_reshape_and_cache_flash,
)
from vllm.v1.kv_cache_interface import KVQuantMode

MAX_NRMSE = 0.18
MIN_COSINE = 0.98
MAX_READER_RATIO = 1.25
HEAD_DIM = 256
NUM_QUERY_HEADS = 6
NUM_KV_HEADS = 1
PAGE_SIZE = 16
TOPK = 2051


def allocate_int4_cache(num_blocks: int):
    shape = Qwen3_8FlashNextQSAFlashAttentionBackend.get_kv_cache_shape(
        num_blocks,
        PAGE_SIZE,
        NUM_KV_HEADS,
        HEAD_DIM,
        "int4_per_token_head",
    )
    raw = torch.zeros(shape, device="cuda", dtype=torch.uint8)
    impl = Qwen3_8FlashNextQSAFlashAttentionImpl.__new__(
        Qwen3_8FlashNextQSAFlashAttentionImpl
    )
    impl.head_size = HEAD_DIM
    impl._kv_quant_mode = KVQuantMode.INT4_PER_TOKEN_HEAD
    key, value, key_scales, value_scales = impl.per_token_head_cache_views(raw)
    return impl, raw, key, value, key_scales, value_scales


def output_metrics(reference: torch.Tensor, actual: torch.Tensor):
    reference = reference.float()
    actual = actual.float()
    delta = actual - reference
    nrmse = torch.sqrt(torch.mean(delta.square())) / torch.clamp(
        torch.sqrt(torch.mean(reference.square())), min=1e-8
    )
    cosine = torch.nn.functional.cosine_similarity(
        reference.flatten(), actual.flatten(), dim=0
    )
    return {
        "normalized_rmse": float(nrmse),
        "cosine_similarity": float(cosine),
        "maximum_absolute_error": float(delta.abs().max()),
    }


def time_operation(operation, iterations=20, repeats=5):
    for _ in range(10):
        operation()
    torch.accelerator.synchronize()
    samples = []
    for _ in range(repeats):
        start = torch.cuda.Event(enable_timing=True)
        end = torch.cuda.Event(enable_timing=True)
        start.record()
        for _ in range(iterations):
            operation()
        end.record()
        end.synchronize()
        samples.append(start.elapsed_time(end) * 1000 / iterations)
    return {
        "median_microseconds": statistics.median(samples),
        "samples": samples,
    }


def performance_case(rows: int):
    torch.manual_seed(1100 + rows)
    num_blocks = (TOPK + PAGE_SIZE - 1) // PAGE_SIZE
    num_tokens = num_blocks * PAGE_SIZE
    key = torch.randn(
        num_tokens,
        NUM_KV_HEADS,
        HEAD_DIM,
        device="cuda",
        dtype=torch.bfloat16,
    )
    value = torch.randn_like(key)
    slots = torch.arange(num_tokens, device="cuda", dtype=torch.int64)
    _, raw, int4_key, int4_value, key_scales, value_scales = allocate_int4_cache(
        num_blocks
    )
    reshape_and_cache_int4(
        key,
        value,
        int4_key,
        int4_value,
        slots,
        k_scale_cache=key_scales,
        v_scale_cache=value_scales,
    )
    bf16_key = key.reshape(num_blocks, PAGE_SIZE, NUM_KV_HEADS, HEAD_DIM).contiguous()
    bf16_value = value.reshape_as(bf16_key).contiguous()
    query = torch.randn(
        rows,
        NUM_QUERY_HEADS,
        HEAD_DIM,
        device="cuda",
        dtype=torch.bfloat16,
    )
    indices = torch.arange(TOPK, device="cuda", dtype=torch.int32).expand(rows, -1)
    block_table = torch.arange(num_blocks, device="cuda", dtype=torch.int32)[None]
    token_to_req = torch.zeros(rows, device="cuda", dtype=torch.int32)

    def bf16_reader():
        return qsa_sparse_paged_attention(
            query,
            bf16_key,
            bf16_value,
            indices,
            block_table,
            token_to_req,
        )

    def int4_reader():
        return qsa_sparse_paged_attention(
            query,
            int4_key,
            int4_value,
            indices,
            block_table,
            token_to_req,
            kv_quant_mode=KVQuantMode.INT4_PER_TOKEN_HEAD,
            key_scale_cache=key_scales,
            value_scale_cache=value_scales,
        )

    bf16_output = bf16_reader()
    int4_output = int4_reader()
    metrics = output_metrics(bf16_output, int4_output)
    bf16_timing = time_operation(bf16_reader)
    int4_timing = time_operation(int4_reader)
    reader_ratio = (
        int4_timing["median_microseconds"] / bf16_timing["median_microseconds"]
    )

    bf16_write_key = torch.empty_like(bf16_key)
    bf16_write_value = torch.empty_like(bf16_value)
    scalar_scale = torch.ones(1, device="cuda", dtype=torch.float32)

    def bf16_writer():
        triton_reshape_and_cache_flash(
            key,
            value,
            bf16_write_key,
            bf16_write_value,
            slots,
            "auto",
            scalar_scale,
            scalar_scale,
        )

    def int4_writer():
        reshape_and_cache_int4(
            key,
            value,
            int4_key,
            int4_value,
            slots,
            k_scale_cache=key_scales,
            v_scale_cache=value_scales,
        )

    bf16_writer_timing = time_operation(bf16_writer)
    int4_writer_timing = time_operation(int4_writer)
    return {
        "rows": rows,
        "output": metrics,
        "numerical_passed": metrics["normalized_rmse"] <= MAX_NRMSE
        and metrics["cosine_similarity"] >= MIN_COSINE,
        "reader": {
            "bf16": bf16_timing,
            "int4": int4_timing,
            "int4_to_bf16_ratio": reader_ratio,
            "passed": reader_ratio <= MAX_READER_RATIO,
        },
        "writer": {
            "bf16": bf16_writer_timing,
            "int4": int4_writer_timing,
            "int4_to_bf16_ratio": int4_writer_timing["median_microseconds"]
            / bf16_writer_timing["median_microseconds"],
        },
    }


def graph_gate():
    torch.manual_seed(1701)
    num_blocks = 4
    num_tokens = num_blocks * PAGE_SIZE
    key = torch.randn(
        num_tokens,
        NUM_KV_HEADS,
        HEAD_DIM,
        device="cuda",
        dtype=torch.bfloat16,
    )
    value = torch.randn_like(key)
    query = torch.randn(
        1,
        NUM_QUERY_HEADS,
        HEAD_DIM,
        device="cuda",
        dtype=torch.bfloat16,
    )
    slots = torch.arange(num_tokens, device="cuda", dtype=torch.int64)
    indices = torch.arange(num_tokens, device="cuda", dtype=torch.int32)[None]
    block_table = torch.arange(num_blocks, device="cuda", dtype=torch.int32)[None]
    token_to_req = torch.zeros(1, device="cuda", dtype=torch.int32)
    _, raw, key_cache, value_cache, key_scales, value_scales = allocate_int4_cache(
        num_blocks
    )

    def transaction():
        reshape_and_cache_int4(
            key,
            value,
            key_cache,
            value_cache,
            slots,
            k_scale_cache=key_scales,
            v_scale_cache=value_scales,
        )
        return qsa_sparse_paged_attention(
            query,
            key_cache,
            value_cache,
            indices,
            block_table,
            token_to_req,
            kv_quant_mode=KVQuantMode.INT4_PER_TOKEN_HEAD,
            key_scale_cache=key_scales,
            value_scale_cache=value_scales,
        )

    for _ in range(3):
        transaction()
    graph = torch.cuda.CUDAGraph()
    with torch.cuda.graph(graph):
        output = transaction()
    graph.replay()
    first = output.clone()
    graph.replay()
    second = output.clone()
    if not torch.equal(first, second) or not torch.isfinite(first).all():
        raise RuntimeError("INT4 QSA CUDA Graph replay is not deterministic and finite")
    return {"bitwise_equal": True, "finite": True}


def main():
    parser = argparse.ArgumentParser()
    parser.add_argument("--output", type=Path, required=True)
    args = parser.parse_args()
    assert torch.cuda.device_count() == 1
    capability = torch.cuda.get_device_capability()
    if capability != (8, 6):
        raise RuntimeError(f"Expected SM86, got {capability}")
    performance = {"m1": performance_case(1), "m256": performance_case(256)}
    result = {
        "schema_version": 1,
        "device": torch.cuda.get_device_name(),
        "compute_capability": list(capability),
        "acceptance": {
            "maximum_normalized_rmse": MAX_NRMSE,
            "minimum_cosine_similarity": MIN_COSINE,
            "maximum_int4_to_bf16_reader_time_ratio": MAX_READER_RATIO,
        },
        "cuda_graph": graph_gate(),
        "performance": performance,
        "numerical_gate_passed": all(
            case["numerical_passed"] for case in performance.values()
        ),
        "performance_gate_passed": all(
            case["reader"]["passed"] for case in performance.values()
        ),
    }
    args.output.parent.mkdir(parents=True, exist_ok=True)
    args.output.write_text(json.dumps(result, indent=2, sort_keys=True) + "\n")
    print(json.dumps(result, indent=2, sort_keys=True))


if __name__ == "__main__":
    main()
