"""Timing-only attribution for QSA INT4 M=1 decode."""

from __future__ import annotations

import json
import statistics

import torch
from vllm.models.qwen3_8_flash_next.nvidia import (
    model as _qwen38_model,  # noqa: F401
)
from vllm.models.qwen3_8_flash_next.nvidia.ops.qsa import (
    qsa_sparse_paged_attention,
)
from vllm.models.qwen3_8_flash_next.nvidia.qsa import (
    Qwen3_8FlashNextQSAFlashAttentionBackend,
    Qwen3_8FlashNextQSAFlashAttentionImpl,
)

from vllm.v1.attention.ops import int4_per_token_head
from vllm.v1.attention.ops.int4_per_token_head import reshape_and_cache_int4
from vllm.v1.kv_cache_interface import KVQuantMode


def time_operation(operation, iterations=100, repeats=7):
    for _ in range(10):
        operation()
    torch.accelerator.synchronize()
    samples = []
    for _ in range(repeats):
        start = torch.cuda.Event(enable_timing=True)
        end = torch.cuda.Event(enable_timing=True)
        start.record()
        for _ in range(iterations):
            operation()
        end.record()
        end.synchronize()
        samples.append(start.elapsed_time(end) * 1000 / iterations)
    return {"median_microseconds": statistics.median(samples), "samples": samples}


def main():
    torch.manual_seed(2301)
    head_dim = 256
    topk = 2051
    page_size = 16
    num_blocks = (topk + page_size - 1) // page_size
    num_tokens = num_blocks * page_size
    key = torch.randn(num_tokens, 1, head_dim, device="cuda", dtype=torch.bfloat16)
    value = torch.randn_like(key)
    query = torch.randn(1, 6, head_dim, device="cuda", dtype=torch.bfloat16)
    shape = Qwen3_8FlashNextQSAFlashAttentionBackend.get_kv_cache_shape(
        num_blocks, page_size, 1, head_dim, "int4_per_token_head"
    )
    raw = torch.zeros(shape, device="cuda", dtype=torch.uint8)
    impl = Qwen3_8FlashNextQSAFlashAttentionImpl.__new__(
        Qwen3_8FlashNextQSAFlashAttentionImpl
    )
    impl.head_size = head_dim
    impl._kv_quant_mode = KVQuantMode.INT4_PER_TOKEN_HEAD
    key_cache, value_cache, key_scales, value_scales = impl.per_token_head_cache_views(
        raw
    )
    reshape_and_cache_int4(
        key,
        value,
        key_cache,
        value_cache,
        torch.arange(num_tokens, device="cuda", dtype=torch.int64),
        k_scale_cache=key_scales,
        v_scale_cache=value_scales,
    )
    indices = torch.arange(topk, device="cuda", dtype=torch.int32)[None]
    block_table = torch.arange(num_blocks, device="cuda", dtype=torch.int32)[None]
    token_to_req = torch.zeros(1, device="cuda", dtype=torch.int32)

    def reader():
        return qsa_sparse_paged_attention(
            query,
            key_cache,
            value_cache,
            indices,
            block_table,
            token_to_req,
            kv_quant_mode=KVQuantMode.INT4_PER_TOKEN_HEAD,
            key_scale_cache=key_scales,
            value_scale_cache=value_scales,
        )

    forward_rht = time_operation(
        lambda: int4_per_token_head.single_rht(query.float()).to(query.dtype)
    )
    transformed = torch.randn_like(query)
    inverse_rht = time_operation(
        lambda: int4_per_token_head.single_rht(transformed.float(), inverse=True).to(
            query.dtype
        )
    )
    full_reader = time_operation(reader)
    original_rht = int4_per_token_head.single_rht
    int4_per_token_head.single_rht = lambda tensor, inverse=False: tensor
    try:
        core_without_rht = time_operation(reader)
    finally:
        int4_per_token_head.single_rht = original_rht
    result = {
        "schema_version": 1,
        "device": torch.cuda.get_device_name(),
        "compute_capability": list(torch.cuda.get_device_capability()),
        "forward_rht": forward_rht,
        "inverse_rht": inverse_rht,
        "full_reader": full_reader,
        "core_without_rht_timing_only": core_without_rht,
    }
    print(json.dumps(result, indent=2, sort_keys=True))


if __name__ == "__main__":
    main()
