import argparse
import importlib.util
import json
import statistics
from pathlib import Path

import torch
import torch.nn.functional as F

MAXIMUM_NRMSE = 0.17
MINIMUM_COSINE = 0.985
MAXIMUM_READER_RATIO = 1.25
HEAD_SIZE = 256
QUERY_HEADS = 6
TOPK = 2051
BLOCK_SIZE = 16


def load_candidate(path: Path):
    name = "vllm.models.qwen3_8_flash_next.nvidia.ops.qsa_q8k_q4v"
    spec = importlib.util.spec_from_file_location(name, path)
    if spec is None or spec.loader is None:
        raise RuntimeError(f"Cannot load candidate module: {path}")
    module = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(module)
    return module


def time_operation(operation, iterations: int, repeats: int = 5) -> dict:
    for _ in range(10):
        operation()
    torch.accelerator.synchronize()
    samples = []
    for _ in range(repeats):
        start = torch.cuda.Event(enable_timing=True)
        end = torch.cuda.Event(enable_timing=True)
        start.record()
        for _ in range(iterations):
            operation()
        end.record()
        end.synchronize()
        samples.append(start.elapsed_time(end) * 1000 / iterations)
    return {"median_microseconds": statistics.median(samples), "samples": samples}


def metrics(reference: torch.Tensor, candidate: torch.Tensor) -> dict:
    reference = reference.float().reshape(-1)
    candidate = candidate.float().reshape(-1)
    error = candidate - reference
    return {
        "normalized_rmse": float(
            torch.linalg.vector_norm(error) / torch.linalg.vector_norm(reference)
        ),
        "cosine_similarity": float(
            F.cosine_similarity(reference[None], candidate[None])
        ),
        "maximum_absolute_error": float(error.abs().amax()),
    }


def graph_gate(operation) -> dict:
    for _ in range(3):
        operation()
    torch.accelerator.synchronize()
    graph = torch.cuda.CUDAGraph()
    with torch.cuda.graph(graph):
        output = operation()
    graph.replay()
    first = output.clone()
    graph.replay()
    second = output.clone()
    torch.accelerator.synchronize()
    return {
        "finite": bool(torch.isfinite(first).all()),
        "bitwise_equal": bool(torch.equal(first, second)),
    }


def main() -> None:
    parser = argparse.ArgumentParser()
    parser.add_argument("--module", type=Path, required=True)
    parser.add_argument("--output", type=Path, required=True)
    args = parser.parse_args()
    if torch.cuda.get_device_capability() != (8, 6):
        raise RuntimeError("QSA mixed-cache gate requires SM86")
    candidate = load_candidate(args.module)
    from vllm.models.qwen3_8_flash_next.nvidia.ops.qsa import (
        qsa_sparse_paged_attention,
    )

    torch.manual_seed(20260829)
    cache_tokens = TOPK
    num_blocks = (cache_tokens + BLOCK_SIZE - 1) // BLOCK_SIZE
    key = torch.randn((cache_tokens, 1, HEAD_SIZE), dtype=torch.bfloat16, device="cuda")
    value = torch.randn_like(key)
    slots = torch.arange(cache_tokens, dtype=torch.int64, device="cuda")
    key_cache = torch.empty(
        (num_blocks, BLOCK_SIZE, 1, HEAD_SIZE), dtype=torch.int8, device="cuda"
    )
    key_scales = torch.empty(
        (num_blocks, BLOCK_SIZE, 1), dtype=torch.float32, device="cuda"
    )
    value_cache = torch.empty(
        (num_blocks, BLOCK_SIZE, 1, HEAD_SIZE // 2),
        dtype=torch.uint8,
        device="cuda",
    )
    value_scales = torch.empty_like(key_scales)

    def mixed_writer():
        candidate.reshape_and_cache_q8k_q4v(
            key,
            value,
            key_cache,
            key_scales,
            value_cache,
            value_scales,
            slots,
        )

    mixed_writer()
    bf16_key_cache = torch.zeros(
        (num_blocks, BLOCK_SIZE, 1, HEAD_SIZE),
        dtype=torch.bfloat16,
        device="cuda",
    )
    bf16_value_cache = torch.zeros_like(bf16_key_cache)
    bf16_key_cache.view(-1, 1, HEAD_SIZE)[:cache_tokens].copy_(key)
    bf16_value_cache.view(-1, 1, HEAD_SIZE)[:cache_tokens].copy_(value)
    block_table = torch.arange(num_blocks, dtype=torch.int32, device="cuda")[None]
    results = {}
    for rows in (1, 256):
        query = torch.randn(
            (rows, QUERY_HEADS, HEAD_SIZE), dtype=torch.bfloat16, device="cuda"
        )
        logical_indices = torch.arange(TOPK, dtype=torch.int32, device="cuda").repeat(
            rows, 1
        )
        token_to_request = torch.zeros(rows, dtype=torch.int32, device="cuda")

        def bf16_reader(
            query=query,
            logical_indices=logical_indices,
            token_to_request=token_to_request,
        ):
            return qsa_sparse_paged_attention(
                query,
                bf16_key_cache,
                bf16_value_cache,
                logical_indices,
                block_table,
                token_to_request,
            )

        def mixed_reader(
            query=query,
            logical_indices=logical_indices,
            token_to_request=token_to_request,
        ):
            return candidate.qsa_sparse_paged_attention_q8k_q4v(
                query,
                key_cache,
                key_scales,
                value_cache,
                value_scales,
                logical_indices,
                block_table,
                token_to_request,
            )

        reference = bf16_reader()
        actual = mixed_reader()
        result = metrics(reference, actual)
        iterations = 100 if rows == 1 else 20
        bf16_timing = time_operation(bf16_reader, iterations)
        mixed_timing = time_operation(mixed_reader, iterations)
        result.update(
            {
                "bf16_timing": bf16_timing,
                "mixed_timing": mixed_timing,
                "reader_ratio": mixed_timing["median_microseconds"]
                / bf16_timing["median_microseconds"],
            }
        )
        if rows == 1:
            result["cuda_graph"] = graph_gate(mixed_reader)
        results[str(rows)] = result

    writer_timing = time_operation(mixed_writer, 20)
    cases = list(results.values())
    report = {
        "schema_version": 1,
        "device": torch.cuda.get_device_name(),
        "compute_capability": list(torch.cuda.get_device_capability()),
        "acceptance": {
            "maximum_normalized_rmse": MAXIMUM_NRMSE,
            "minimum_cosine_similarity": MINIMUM_COSINE,
            "maximum_reader_ratio": MAXIMUM_READER_RATIO,
        },
        "reader": results,
        "writer": writer_timing,
        "numerical_passed": all(
            case["normalized_rmse"] <= MAXIMUM_NRMSE
            and case["cosine_similarity"] >= MINIMUM_COSINE
            for case in cases
        ),
        "performance_passed": all(
            case["reader_ratio"] <= MAXIMUM_READER_RATIO for case in cases
        ),
        "graph_passed": all(results["1"]["cuda_graph"].values()),
    }
    args.output.write_text(json.dumps(report, indent=2, sort_keys=True) + "\n")
    print(
        json.dumps(
            {
                key: report[key]
                for key in ("numerical_passed", "performance_passed", "graph_passed")
            },
            sort_keys=True,
        )
    )
    if not all(
        report[key]
        for key in ("numerical_passed", "performance_passed", "graph_passed")
    ):
        raise SystemExit(1)


if __name__ == "__main__":
    main()
