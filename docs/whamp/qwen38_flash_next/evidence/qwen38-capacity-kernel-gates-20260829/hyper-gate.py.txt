import argparse
import importlib.util
import json
import statistics
from functools import partial
from pathlib import Path

import torch
import torch.nn.functional as F
from safetensors import safe_open

MAXIMUM_NRMSE = 0.02
MINIMUM_COSINE = 0.9999
MINIMUM_SPEED_RATIO = 0.90


def load_module(path: Path):
    spec = importlib.util.spec_from_file_location("qwen_hc_int8_candidate", path)
    if spec is None or spec.loader is None:
        raise RuntimeError(f"Cannot load candidate module: {path}")
    module = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(module)
    return module


def load_weights(model: Path) -> dict[str, torch.Tensor]:
    index = json.loads((model / "model.safetensors.index.json").read_text())
    prefix = "model.language_model.layers.0.attn_hyper_connection."
    names = {
        "down": prefix + "input_mix_weight_down.weight",
        "inject": prefix + "block_inject_weight.weight",
        "up": prefix + "input_mix_weight_up.weight",
    }
    shards = {index["weight_map"][name] for name in names.values()}
    if len(shards) != 1:
        raise RuntimeError("Layer-0 hyperconnection weights cross source shards")
    with safe_open(model / next(iter(shards)), framework="pt", device="cpu") as source:
        loaded = {kind: source.get_tensor(name) for kind, name in names.items()}
    valid_rows = loaded["down"].shape[0] + loaded["inject"].shape[0]
    padding_rows = (-valid_rows) % 16
    merged = torch.cat(
        (
            loaded["down"],
            loaded["inject"],
            torch.zeros(
                (padding_rows, loaded["down"].shape[1]),
                dtype=loaded["down"].dtype,
            ),
        )
    )
    return {"merged_down": merged, "up": loaded["up"]}


def quantize_weight(
    weight: torch.Tensor,
    group_size: int,
    valid_rows: int | None,
) -> tuple[torch.Tensor, torch.Tensor]:
    rows, columns = weight.shape
    valid_rows = rows if valid_rows is None else valid_rows
    grouped = (
        weight[:valid_rows]
        .float()
        .reshape(valid_rows, columns // group_size, group_size)
    )
    scales = (grouped.abs().amax(dim=2) / 127).clamp_min(1e-8).to(torch.float16)
    codes = (
        (grouped / scales.float()[:, :, None])
        .round()
        .clamp(-127, 127)
        .to(torch.int8)
        .reshape(valid_rows, columns)
    )
    output_codes = torch.zeros((rows, columns), dtype=torch.int8, device=weight.device)
    output_scales = torch.ones(
        (rows, columns // group_size), dtype=torch.float16, device=weight.device
    )
    output_codes[:valid_rows] = codes
    output_scales[:valid_rows] = scales
    return output_codes, output_scales


def metrics(reference: torch.Tensor, candidate: torch.Tensor) -> dict:
    reference = reference.float().reshape(-1)
    candidate = candidate.float().reshape(-1)
    error = candidate - reference
    return {
        "normalized_rmse": float(
            torch.linalg.vector_norm(error) / torch.linalg.vector_norm(reference)
        ),
        "cosine_similarity": float(
            F.cosine_similarity(reference[None], candidate[None])
        ),
        "maximum_absolute_error": float(error.abs().amax()),
    }


def time_operation(operation, iterations: int, repeats: int = 5) -> dict:
    for _ in range(20):
        operation()
    torch.accelerator.synchronize()
    samples = []
    for _ in range(repeats):
        start = torch.cuda.Event(enable_timing=True)
        end = torch.cuda.Event(enable_timing=True)
        start.record()
        for _ in range(iterations):
            operation()
        end.record()
        end.synchronize()
        samples.append(start.elapsed_time(end) * 1000 / iterations)
    return {"median_microseconds": statistics.median(samples), "samples": samples}


def graph_gate(module, inputs, codes, scales, group_size) -> dict:
    for _ in range(3):
        module.hyperconnection_int8_decode(inputs, codes, scales, group_size)
    torch.accelerator.synchronize()
    graph = torch.cuda.CUDAGraph()
    with torch.cuda.graph(graph):
        output = module.hyperconnection_int8_decode(inputs, codes, scales, group_size)
    graph.replay()
    first = output.clone()
    graph.replay()
    second = output.clone()
    torch.accelerator.synchronize()
    return {
        "bitwise_equal": bool(torch.equal(first, second)),
        "finite": bool(torch.isfinite(first).all()),
    }


def main() -> None:
    parser = argparse.ArgumentParser()
    parser.add_argument("--module", type=Path, required=True)
    parser.add_argument("--model", type=Path, required=True)
    parser.add_argument("--output", type=Path, required=True)
    args = parser.parse_args()
    if torch.cuda.get_device_capability() != (8, 6):
        raise RuntimeError("Hyperconnection gate requires SM86")
    module = load_module(args.module)
    torch.manual_seed(20260829)
    results = {}
    for name, source_cpu in load_weights(args.model).items():
        source = source_cpu.cuda()
        group_size = 128 if name == "merged_down" else source.shape[1]
        valid_rows = 324 if name == "merged_down" else None
        codes, scales = quantize_weight(source, group_size, valid_rows)
        dequantized = torch.empty_like(source)
        module.dequantize_hyperconnection_int8_weight(
            codes, scales, dequantized, group_size
        )
        matrix_result = {
            "shape": list(source.shape),
            "group_size": group_size,
            "storage_bytes": codes.nbytes + scales.nbytes,
            "bf16_bytes": source.nbytes,
            "token_counts": {},
        }
        for token_count in (1, 2, 256):
            inputs = torch.randn(
                (token_count, source.shape[1]),
                dtype=torch.bfloat16,
                device="cuda",
            )
            reference = F.linear(inputs, source)
            if token_count <= 2:
                operation = partial(
                    module.hyperconnection_int8_decode,
                    inputs,
                    codes,
                    scales,
                    group_size,
                )
            else:

                def operation(
                    inputs=inputs,
                    codes=codes,
                    scales=scales,
                    dequantized=dequantized,
                    group_size=group_size,
                ):
                    module.dequantize_hyperconnection_int8_weight(
                        codes, scales, dequantized, group_size
                    )
                    return F.linear(inputs, dequantized)

            candidate = operation()
            result = metrics(reference, candidate)
            iterations = 1000 if token_count <= 2 else 100
            bf16_timing = time_operation(
                lambda inputs=inputs, source=source: F.linear(inputs, source),
                iterations,
            )
            candidate_timing = time_operation(operation, iterations)
            result.update(
                {
                    "bf16_timing": bf16_timing,
                    "candidate_timing": candidate_timing,
                    "speed_ratio": bf16_timing["median_microseconds"]
                    / candidate_timing["median_microseconds"],
                }
            )
            if token_count <= 2:
                result["cuda_graph"] = graph_gate(
                    module, inputs, codes, scales, group_size
                )
            matrix_result["token_counts"][str(token_count)] = result
        results[name] = matrix_result
    cases = [
        case for matrix in results.values() for case in matrix["token_counts"].values()
    ]
    report = {
        "schema_version": 1,
        "device": torch.cuda.get_device_name(),
        "compute_capability": list(torch.cuda.get_device_capability()),
        "acceptance": {
            "maximum_normalized_rmse": MAXIMUM_NRMSE,
            "minimum_cosine_similarity": MINIMUM_COSINE,
            "minimum_speed_ratio": MINIMUM_SPEED_RATIO,
        },
        "matrices": results,
        "numerical_passed": all(
            case["normalized_rmse"] <= MAXIMUM_NRMSE
            and case["cosine_similarity"] >= MINIMUM_COSINE
            for case in cases
        ),
        "performance_passed": all(
            case["speed_ratio"] >= MINIMUM_SPEED_RATIO for case in cases
        ),
        "graph_passed": all(
            all(
                case.get("cuda_graph", {"finite": True, "bitwise_equal": True}).values()
            )
            for case in cases
        ),
    }
    args.output.write_text(json.dumps(report, indent=2, sort_keys=True) + "\n")
    print(
        json.dumps(
            {
                key: report[key]
                for key in ("numerical_passed", "performance_passed", "graph_passed")
            },
            sort_keys=True,
        )
    )
    if not all(
        report[key]
        for key in ("numerical_passed", "performance_passed", "graph_passed")
    ):
        raise SystemExit(1)


if __name__ == "__main__":
    main()
