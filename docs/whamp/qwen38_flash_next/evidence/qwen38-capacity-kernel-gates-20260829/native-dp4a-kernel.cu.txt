#include <cuda_bf16.h>
#include <cuda_fp16.h>
#include <cuda_runtime.h>

#include <cstdint>

namespace vllm::qwen4_exp {
namespace {

constexpr int kWarpSize = 32;
constexpr int kWarpsPerBlock = 8;

__device__ __forceinline__ float warp_max(float value) {
#pragma unroll
  for (int offset = 16; offset > 0; offset >>= 1) {
    value = fmaxf(value, __shfl_xor_sync(0xffffffff, value, offset));
  }
  return value;
}

__device__ __forceinline__ int warp_sum(int value) {
#pragma unroll
  for (int offset = 16; offset > 0; offset >>= 1) {
    value += __shfl_down_sync(0xffffffff, value, offset);
  }
  return value;
}

__device__ __forceinline__ int8_t quantize_int8(float value, float scale) {
  const int rounded = __float2int_rn(value / scale);
  return static_cast<int8_t>(max(-127, min(127, rounded)));
}

template <int kGroupSize>
__global__ void hyperconnection_int8_decode_kernel(
    const __nv_bfloat16* __restrict__ input, const int8_t* __restrict__ weight,
    const __half* __restrict__ weight_scale, __nv_bfloat16* __restrict__ output,
    int token_count, int output_rows, int input_columns) {
  static_assert(kGroupSize % kWarpSize == 0);
  constexpr int kValuesPerLane = kGroupSize / kWarpSize;
  constexpr int kPacksPerLane = (kValuesPerLane + 3) / 4;

  const int lane = threadIdx.x & (kWarpSize - 1);
  const int warp = threadIdx.x / kWarpSize;
  const int output_index = blockIdx.x * kWarpsPerBlock + warp;
  const int output_count = token_count * output_rows;
  if (output_index >= output_count) {
    return;
  }

  const int token_index = output_index / output_rows;
  const int output_row = output_index % output_rows;
  const int group_count = input_columns / kGroupSize;
  float accumulator = 0.0f;

  for (int group = 0; group < group_count; ++group) {
    float values[kValuesPerLane];
    float absolute_maximum = 0.0f;
#pragma unroll
    for (int index = 0; index < kValuesPerLane; ++index) {
      const int column = group * kGroupSize + lane + index * kWarpSize;
      values[index] =
          __bfloat162float(input[token_index * input_columns + column]);
      absolute_maximum = fmaxf(absolute_maximum, fabsf(values[index]));
    }
    absolute_maximum = warp_max(absolute_maximum);
    const float activation_scale =
        absolute_maximum == 0.0f ? 1.0f : absolute_maximum / 127.0f;

    int integer_accumulator = 0;
#pragma unroll
    for (int pack_index = 0; pack_index < kPacksPerLane; ++pack_index) {
      uint32_t activation_pack = 0;
      uint32_t weight_pack = 0;
#pragma unroll
      for (int byte_index = 0; byte_index < 4; ++byte_index) {
        const int value_index = pack_index * 4 + byte_index;
        if (value_index < kValuesPerLane) {
          const int column =
              group * kGroupSize + lane + value_index * kWarpSize;
          const int8_t activation_code =
              quantize_int8(values[value_index], activation_scale);
          const int8_t weight_code =
              weight[output_row * input_columns + column];
          activation_pack |=
              static_cast<uint32_t>(static_cast<uint8_t>(activation_code))
              << (byte_index * 8);
          weight_pack |=
              static_cast<uint32_t>(static_cast<uint8_t>(weight_code))
              << (byte_index * 8);
        }
      }
      integer_accumulator =
          __dp4a(static_cast<int>(activation_pack),
                 static_cast<int>(weight_pack), integer_accumulator);
    }
    integer_accumulator = warp_sum(integer_accumulator);
    if (lane == 0) {
      const float scale =
          __half2float(weight_scale[output_row * group_count + group]);
      accumulator = fmaf(static_cast<float>(integer_accumulator),
                         activation_scale * scale, accumulator);
    }
  }

  if (lane == 0) {
    output[output_index] = __float2bfloat16_rn(accumulator);
  }
}

__global__ void dequantize_hyperconnection_int8_weight_kernel(
    const int8_t* __restrict__ weight, const __half* __restrict__ weight_scale,
    __nv_bfloat16* __restrict__ output, int element_count, int input_columns,
    int group_size, int group_count) {
  const int first_element = (blockIdx.x * blockDim.x + threadIdx.x) * 4;
  if (first_element >= element_count) {
    return;
  }

  const int output_row = first_element / input_columns;
  const int input_column = first_element % input_columns;
  const int group = input_column / group_size;
  const float scale =
      __half2float(weight_scale[output_row * group_count + group]);

  if (first_element + 3 < element_count && input_column + 3 < input_columns) {
    const int packed = *reinterpret_cast<const int*>(weight + first_element);
    const auto code0 = static_cast<int8_t>(packed & 0xff);
    const auto code1 = static_cast<int8_t>((packed >> 8) & 0xff);
    const auto code2 = static_cast<int8_t>((packed >> 16) & 0xff);
    const auto code3 = static_cast<int8_t>((packed >> 24) & 0xff);
    auto* output_pairs =
        reinterpret_cast<__nv_bfloat162*>(output + first_element);
    output_pairs[0] = __floats2bfloat162_rn(code0 * scale, code1 * scale);
    output_pairs[1] = __floats2bfloat162_rn(code2 * scale, code3 * scale);
    return;
  }

  for (int offset = 0; offset < 4 && first_element + offset < element_count;
       ++offset) {
    output[first_element + offset] = __float2bfloat16_rn(
        static_cast<float>(weight[first_element + offset]) * scale);
  }
}

extern "C" int qwen4_exp_hyperconnection_int8_decode_sm86(
    uintptr_t input, uintptr_t weight, uintptr_t weight_scale, uintptr_t output,
    int token_count, int output_rows, int input_columns, int group_size,
    uintptr_t stream) {
  if (token_count < 1 || token_count > 2 || output_rows <= 0 ||
      input_columns <= 0 || input_columns % group_size != 0) {
    return static_cast<int>(cudaErrorInvalidValue);
  }
  const dim3 block(256);
  const dim3 grid((token_count * output_rows + 7) / 8);
  auto* input_ptr = reinterpret_cast<const __nv_bfloat16*>(input);
  auto* weight_ptr = reinterpret_cast<const int8_t*>(weight);
  auto* scale_ptr = reinterpret_cast<const __half*>(weight_scale);
  auto* output_ptr = reinterpret_cast<__nv_bfloat16*>(output);
  const auto cuda_stream = reinterpret_cast<cudaStream_t>(stream);
  if (group_size == 128) {
    vllm::qwen4_exp::hyperconnection_int8_decode_kernel<128>
        <<<grid, block, 0, cuda_stream>>>(input_ptr, weight_ptr, scale_ptr,
                                          output_ptr, token_count, output_rows,
                                          input_columns);
  } else if (group_size == 320) {
    vllm::qwen4_exp::hyperconnection_int8_decode_kernel<320>
        <<<grid, block, 0, cuda_stream>>>(input_ptr, weight_ptr, scale_ptr,
                                          output_ptr, token_count, output_rows,
                                          input_columns);
  } else {
    return static_cast<int>(cudaErrorInvalidValue);
  }
  return static_cast<int>(cudaGetLastError());
}

extern "C" int qwen4_exp_dequantize_hyperconnection_int8_weight_sm86(
    uintptr_t weight, uintptr_t weight_scale, uintptr_t output, int output_rows,
    int input_columns, int group_size, uintptr_t stream) {
  if (output_rows <= 0 || input_columns <= 0 || group_size <= 0 ||
      input_columns % group_size != 0) {
    return static_cast<int>(cudaErrorInvalidValue);
  }
  constexpr int kElementsPerThread = 4;
  constexpr int kThreads = 256;
  const int element_count = output_rows * input_columns;
  const int group_count = input_columns / group_size;
  const int blocks = (element_count + kElementsPerThread * kThreads - 1) /
                     (kElementsPerThread * kThreads);
  const auto cuda_stream = reinterpret_cast<cudaStream_t>(stream);
  vllm::qwen4_exp::dequantize_hyperconnection_int8_weight_kernel<<<
      blocks, kThreads, 0, cuda_stream>>>(
      reinterpret_cast<const int8_t*>(weight),
      reinterpret_cast<const __half*>(weight_scale),
      reinterpret_cast<__nv_bfloat16*>(output), element_count, input_columns,
      group_size, group_count);
  return static_cast<int>(cudaGetLastError());
}
