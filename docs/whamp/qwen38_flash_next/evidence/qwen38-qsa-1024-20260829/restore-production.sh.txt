#!/usr/bin/env bash
set -euo pipefail

candidate_compose=/home/will/inference/runtime/qwen38-qsa-1024/candidate.yml
production_container=qwen38-flash-next-intel-autoround-vllm
production_image=sha256:0aea30240f3e3d9ffae8526643950e170eb5fa07fc427016a9dd90892afa2aa3
production_url=http://127.0.0.1:30001

restore_overcommit() {
  sudo -n sysctl -q -w vm.overcommit_memory=0 >/dev/null || true
}
trap restore_overcommit EXIT

if [[ -f "$candidate_compose" ]]; then
  docker compose -p qwen38-qsa-1024 --profile qwen38-flash-next \
    -f "$candidate_compose" down --remove-orphans || true
fi
sudo -n swapoff -a
sudo -n sysctl -q -w vm.overcommit_memory=1 >/dev/null
if docker inspect "$production_container" >/dev/null 2>&1; then
  docker start "$production_container" >/dev/null
else
  cd /home/will/inference/serving/vllm
  docker compose -p vllm --profile qwen38-flash-next \
    -f compose.qwen38-flash-next-intel.yaml up -d
fi

health_deadline=$((SECONDS + 3600))
while (( SECONDS < health_deadline )); do
  if curl -fsS "$production_url/health" >/dev/null 2>&1; then
    break
  fi
  if [[ "$(docker inspect -f '{{.State.Running}}' "$production_container" 2>/dev/null || true)" != true ]]; then
    docker logs --tail 300 "$production_container" >&2 || true
    exit 1
  fi
  sleep 15
done
if (( SECONDS >= health_deadline )); then
  echo "Production service did not become healthy within 60 minutes" >&2
  docker logs --tail 300 "$production_container" >&2 || true
  exit 1
fi

restore_overcommit
trap - EXIT

actual_image=$(docker inspect -f '{{.Image}}' "$production_container")
[[ "$actual_image" == "$production_image" ]]
[[ "$(docker inspect -f '{{.RestartCount}}' "$production_container")" == 0 ]]
[[ "$(cat /proc/swaps | wc -l)" == 1 ]]
[[ "$(sysctl -n vm.overcommit_memory)" == 0 ]]
[[ "$(systemctl is-active gpu-power-limit.service)" == active ]]
for pid in $(docker top "$production_container" -o pid | tail -n +2); do
  swap_kib=$(awk '/^VmSwap:/{print $2}' "/proc/$pid/status" 2>/dev/null || echo 0)
  [[ "${swap_kib:-0}" == 0 ]]
done
curl -fsS "$production_url/v1/models" | grep -q 'qwen3.8-flash-next-intel-autoround-w4a16'
response=$(curl -fsS "$production_url/v1/chat/completions" \
  -H 'Content-Type: application/json' \
  -d '{"model":"qwen3.8-flash-next-intel-autoround-w4a16","messages":[{"role":"user","content":"Reply with exactly PARIS"}],"temperature":0,"max_tokens":128}')
python3 -c 'import json,sys; r=json.load(sys.stdin); assert r["choices"][0]["message"]["content"].strip()=="PARIS"; assert r["choices"][0]["finish_reason"]=="stop"' <<<"$response"
echo 'Production Qwen3.8 service restored healthy with zero swap.'
